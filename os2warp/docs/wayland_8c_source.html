35/* 11x16 arrow, hotspot at the tip (top-left, row/col 0). Bit set = cursor
36 * pixel. Same shape convention as the classic Windows/OS2 arrow pointer. */
37static const uint16_t cursor_shape[16] = {
38 0x8000, 0xC000, 0xE000, 0xF000,
39 0xF800, 0xFC00, 0xFE00, 0xFF00,
40 0xFF80, 0xFFE0, 0xFC00, 0xDC00,
41 0x8600, 0x0600, 0x0300, 0x0000,

44static void draw_cursor(void)
45{
46 for (int y = 0; y < 16; y++) {
47 uint16_t row = cursor_shape[y];
48 for (int x = 0; x < 11; x++) {
49 if (!(row & (0x8000u >> x))) continue;
50 fb_putpixel(cursor_x + x, cursor_y + y, WPS_TEXT_WHITE);
51 /* Black outline on the trailing edge of each filled run. */
52 if (x == 10 || !(row & (0x8000u >> (x + 1)))) {
53 fb_putpixel(cursor_x + x + 1, cursor_y + y, WPS_TEXT_BLACK);
54 }
55 }
56 }

78wl_surface_t *wl_compositor_create_surface(int width, int height)
79{
80 if (surface_count >= MAX_SURFACES) {
81 kprintf("[WL] ERROR: Too many surfaces (max %d)\r\n", MAX_SURFACES);
82 return NULL;
83 }
84 if (width <= 0 || height <= 0) return NULL;
85
86 wl_surface_t *s = (wl_surface_t *)mem_alloc(sizeof(wl_surface_t));
87 if (!s) {
88 kprintf("[WL] ERROR: Failed to allocate surface\r\n");
89 return NULL;
90 }
91 memset(s, 0, sizeof(*s));
92
93 s->buffer = (fb_color_t *)mem_alloc((size_t)width * (size_t)height * sizeof(fb_color_t));
94 if (!s->buffer) {
95 kprintf("[WL] ERROR: Failed to allocate surface buffer\r\n");
96 mem_free(s);
97 return NULL;
98 }
99 memset(s->buffer, 0, (size_t)width * (size_t)height * sizeof(fb_color_t));
100
101 s->width = width;
102 s->height = height;
103 s->role = WL_SURFACE_ROLE_NONE;
104 s->workplace = active_workplace;
105 s->mapped = true;
106 s->has_damage = true; /* new surfaces need an initial paint */
107 s->damage = (fb_rect_t){0, 0, width, height};
108
109 surfaces[surface_count++] = s;
110 return s;
111}

113void wl_surface_destroy(wl_surface_t *surface)
114{
115 if (!surface) return;
116
117 for (int i = 0; i < surface_count; i++) {
118 if (surfaces[i] == surface) {
119 for (int j = i; j < surface_count - 1; j++) surfaces[j] = surfaces[j + 1];
120 surface_count--;
121 surfaces[surface_count] = NULL;
122
123 if (focused_surface == surface) {
124 focused_surface = surface_count > 0 ? surfaces[surface_count - 1] : NULL;
125 }
126 if (dragging_surface == surface) dragging_surface = NULL;
127
128 if (surface->buffer) mem_free(surface->buffer);
129 mem_free(surface);
130 return;
131 }
132 }

149 focused_surface = surface;

169void wl_surface_damage(wl_surface_t *surface, int x, int y, int w, int h)
170{
171 if (!surface) return;
172
173 if (!surface->has_damage) {
174 surface->damage = (fb_rect_t){x, y, w, h};
175 } else {
176 /* Union the new region into the accumulated damage rect. */
177 int x0 = surface->damage.x < x ? surface->damage.x : x;
178 int y0 = surface->damage.y < y ? surface->damage.y : y;
179 int x1a = surface->damage.x + surface->damage.width;
180 int y1a = surface->damage.y + surface->damage.height;
181 int x1b = x + w;
182 int y1b = y + h;
183 int x1 = x1a > x1b ? x1a : x1b;
184 int y1 = y1a > y1b ? y1a : y1b;
185 surface->damage = (fb_rect_t){x0, y0, x1 - x0, y1 - y0};
186 }
187 surface->has_damage = true;
188}

252void wl_buf_drawstring(wl_surface_t *surface, int x, int y, const char *str, fb_color_t fg, fb_color_t bg)
253{
254 if (!surface) return;
255 int cx = x;
256 int cy = y;
257
258 while (*str) {
259 const uint8_t *glyph = fb_get_glyph(*str);
260 for (int row = 0; row < 8; row++) {
261 uint8_t bits = glyph[row];
262 for (int col = 0; col < 8; col++) {
263 fb_color_t c = (bits & (1 << (7 - col))) ? fg : bg;
264 wl_buf_putpixel(surface, cx + col, cy + row, c);
265 }
266 }
267 cx += 8;
268 str++;
269 }

306void wl_compositor_flush(void)
307{
308 for (int i = 0; i < surface_count; i++) {
309 wl_surface_t *s = surfaces[i];
310 if (!s->mapped) continue;
311 if (s->workplace != active_workplace) continue;
312
313 wl_surface_commit(s);
314
315 int client_x = s->x;
316 int client_y = s->y;
317 if (s->role == WL_SURFACE_ROLE_TOPLEVEL) {
318 draw_decorations(s);
319 client_y += TITLE_BAR_HEIGHT;
320 }
321 blit_surface(s, client_x, client_y);
322 }
323
324 /* Cursor last, on top of every surface. */
325 draw_cursor();
326
327 /* Everything above drew into the back buffer; copy it to the front
328 * buffer so it's actually visible (ramfb, if attached, only ever
329 * displays the front buffer - see gui/ramfb.h). */
330 fb_present();
331}

340void wl_compositor_set_workplace(int index)
341{
342 if (index < 0 || index >= WL_MAX_WORKPLACES) return;
343 if (index == active_workplace) return;
344
345 active_workplace = index;
346 kprintf("[WL] Switched to workplace %d\r\n", active_workplace + 1);
347
348 /* wl_compositor_flush() only ever blits mapped surfaces on top of
349 * whatever's already in the back buffer - it never clears it - so
350 * the outgoing workplace's content would otherwise stay visible
351 * around/behind the incoming one. Clear, then fully redamage every
352 * surface that belongs here now before compositing it in. */
353 fb_clear(WPS_DESKTOP_BG);
354 for (int i = 0; i < surface_count; i++) {
355 wl_surface_t *s = surfaces[i];
356 if (s->workplace != active_workplace) continue;
357 s->has_damage = true;
358 s->damage = (fb_rect_t){0, 0, s->width, s->height};
359 }
360
361 wl_compositor_flush();
362}

371void wl_seat_pointer_button(int x, int y, int button, bool pressed)
372{
373 (void)button;
374 kprintf("[WL] Pointer button %d %s at (%d,%d)\r\n",
375 button, pressed ? "down" : "up", x, y);
376 if (!pressed) {
377 dragging_surface = NULL;
378 return;
379 }
380
381 for (int i = surface_count - 1; i >= 0; i--) {
382 wl_surface_t *s = surfaces[i];
383 if (!s->mapped) continue;
384
385 int deco_h = (s->role == WL_SURFACE_ROLE_TOPLEVEL) ? TITLE_BAR_HEIGHT : 0;
386 if (x >= s->x && x < s->x + s->width &&
387 y >= s->y && y < s->y + s->height + deco_h)
388 {
389 if (s->role == WL_SURFACE_ROLE_TOPLEVEL) {
390 int close_x = s->x + s->width - 20;
391 int close_y = s->y + 4;
392 if (x >= close_x && x < close_x + 16 && y >= close_y && y < close_y + 16) {
393 xdg_toplevel_close(s);
394 return;
395 }
396 if (y >= s->y && y < s->y + TITLE_BAR_HEIGHT) {
397 dragging_surface = s;
398 drag_offset_x = x - s->x;
399 drag_offset_y = y - s->y;
400 }
401 }
402
403 wl_compositor_raise(s);
404 return;
405 }
406 }
