71void ai_record_command(const char *name)
72{
73 int idx = -1;
74 for (int i = 0; i < NUM_KNOWN_COMMANDS; i++) {
75 if (strcmp(name, known_commands[i].name) == 0) {
76 idx = i;
77 break;
78 }
79 }
80 if (idx < 0)
81 return;
82
83 const char *canonical = known_commands[idx].name;
84 total_commands++;
85
86 for (int i = 0; i < tracked_count; i++) {
87 if (tracked[i].name == canonical) {
88 tracked[i].count++;
89 return;
90 }
91 }
92
93 if (tracked_count < MAX_TRACKED) {
94 tracked[tracked_count].name = canonical;
95 tracked[tracked_count].count = 1;
96 tracked_count++;
97 }
98}

100static uint32_t tracked_count_for(const char *name)
101{
102 for (int i = 0; i < tracked_count; i++) {
103 if (strcmp(tracked[i].name, name) == 0)
104 return tracked[i].count;
105 }
106 return 0;

109/* Picks the next thing worth telling the user: an untried command first
110 * (in known_commands order), then a round-robin generic tip. */
111static const char *next_tip(void)
112{
113 for (int i = 0; i < NUM_KNOWN_COMMANDS; i++) {
114 if (tracked_count_for(known_commands[i].name) == 0)
115 return known_commands[i].tip;
116 }
117
118 const char *tip = generic_tips[next_generic_tip];
119 next_generic_tip = (next_generic_tip + 1) % NUM_GENERIC_TIPS;
120 return tip;
121}

162 kprintf(" Heap usage: %u%% (%u of %u bytes)\r\n",

175static void ai_stats(void)
176{
177 uart_puts("\r\n");
178 if (tracked_count == 0) {
179 uart_puts("No commands tracked yet this session.\r\n\r\n");
180 return;
181 }
182
183 /* Small N (<= MAX_TRACKED) - a plain insertion sort by count is fine. */
184 tracked_cmd_t sorted[MAX_TRACKED];
185 memcpy(sorted, tracked, sizeof(tracked_cmd_t) * (size_t)tracked_count);
186 for (int i = 1; i < tracked_count; i++) {
187 tracked_cmd_t key = sorted[i];
188 int j = i - 1;
189 while (j >= 0 && sorted[j].count < key.count) {
190 sorted[j + 1] = sorted[j];
191 j--;
192 }
193 sorted[j + 1] = key;
194 }
195
196 uart_puts("Command usage this session:\r\n\r\n");
197 for (int i = 0; i < tracked_count; i++) {
198 uart_puts(" ");
199 uart_puts(sorted[i].name);
200 for (size_t p = strlen(sorted[i].name); p < 8; p++)
201 uart_putc(' ');
202 kprintf("%u\r\n", sorted[i].count);
203 }
204 uart_puts("\r\n");
205}
