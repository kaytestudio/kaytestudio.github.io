25static int stricmp(const char *a, const char *b)
26{
27 while (*a && *b) {
28 char ca = *a, cb = *b;
29 if (ca >= 'a' && ca <= 'z') ca -= 32;
30 if (cb >= 'a' && cb <= 'z') cb -= 32;
31 if (ca != cb) return ca - cb;
32 a++; b++;
33 }
34 return *a - *b;
35}

55static int parse_ifs_line(const char *line, char *name, char *params)
56{
57 // Skip "IFS="
58 const char *p = line;
59 if (p[0] == 'I' && p[1] == 'F' && p[2] == 'S' && p[3] == '=') {
60 p += 4;
61 } else {
62 return -1; // Not an IFS line
63 }
64
65 p = skip_whitespace(p);
66
67 // Extract driver name (up to .IFS)
68 int i = 0;
69 while (*p && *p != '.' && *p != ' ' && *p != '\t' && i < 31) {
70 name[i++] = *p++;
71 }
72 name[i] = '\0';
73
74 if (i == 0) return -1; // No name
75
76 // Skip .IFS extension
77 if (*p == '.') {
78 p++;
79 if (*p == 'I' && p[1] == 'F' && p[2] == 'S') {
80 p += 3;
81 }
82 }
83
84 // Extract parameters
85 p = skip_whitespace(p);
86 i = 0;
87 while (*p && *p != '\r' && *p != '\n' && i < 255) {
88 params[i++] = *p++;
89 }
90 params[i] = '\0';
91
92 return 0;
93}

136int ifs_load_from_line(const char *line)
137{
138 char name[32];
139 char params[256];
140
141 // Parse IFS line
142 if (parse_ifs_line(line, name, params) != 0) {
143 return -1; // Not an IFS line or parse error
144 }
145
146 kprintf("[IFS] Loading driver: %s", name);
147 if (params[0]) {
148 kprintf(" (params: %s)", params);
149 }
150 kprintf("\n");
151
152 // Find driver in registry
153 ifs_driver_t *drv = NULL;
154 for (int i = 0; i < ifs_count; i++) {
155 if (stricmp(ifs_drivers[i].name, name) == 0) {
156 drv = &ifs_drivers[i];
157 break;
158 }
159 }
160
161 if (!drv) {
162 kprintf("[IFS] ERROR: Driver '%s' not found\n", name);
163 kprintf("[IFS] Available drivers:\n");
164 for (int i = 0; i < ifs_count; i++) {
165 kprintf("[IFS] - %s\n", ifs_drivers[i].name);
166 }
167 return -1;
168 }
169
170 // Initialize driver
171 int ret = drv->init(params);
172 if (ret != 0) {
173 kprintf("[IFS] ERROR: Driver '%s' initialization failed: %d\n", name, ret);
174 return -1;
175 }
176
177 kprintf("[IFS] Driver '%s' loaded successfully\n", name);
178 return 0;
179}

181int ifs_load_from_config(const char *config_path)
182{
183 kprintf("[IFS] Reading CONFIG.SYS: %s\n", config_path);
184
185 // Open CONFIG.SYS
186 int fd = vfs_open(config_path, O_RDONLY);
187 if (fd < 0) {
188 kprintf("[IFS] WARNING: Could not open %s\n", config_path);
189 return 0; // Not a fatal error
190 }
191
192 // Read CONFIG.SYS line by line
193 char line[512];
194 int line_pos = 0;
195 int drivers_loaded = 0;
196
197 while (1) {
198 char c;
199 int bytes = vfs_read(fd, &c, 1);
200
201 if (bytes <= 0) break; // EOF or error
202
203 if (c == '\n' || c == '\r') {
204 // End of line
205 if (line_pos > 0) {
206 line[line_pos] = '\0';
207
208 // Check if IFS line
209 if (line[0] == 'I' && line[1] == 'F' && line[2] == 'S' && line[3] == '=') {
210 if (ifs_load_from_line(line) == 0) {
211 drivers_loaded++;
212 }
213 }
214
215 line_pos = 0;
216 }
217 } else {
218 // Add to line buffer
219 if (line_pos < sizeof(line) - 1) {
220 line[line_pos++] = c;
221 }
222 }
223 }
224
225 // Process last line if no newline at EOF
226 if (line_pos > 0) {
227 line[line_pos] = '\0';
228 if (line[0] == 'I' && line[1] == 'F' && line[2] == 'S' && line[3] == '=') {
229 if (ifs_load_from_line(line) == 0) {
230 drivers_loaded++;
231 }
232 }
233 }
234
235 vfs_close(fd);
236
237 kprintf("[IFS] Loaded %d driver(s) from CONFIG.SYS\n", drivers_loaded);
238 return drivers_loaded;
239}
