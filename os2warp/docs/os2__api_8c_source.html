47void DosResolvePath(const char *in, char *out, uint32_t out_size)
48{
49 char tmp[VFS_PATH_MAX];
50 uint32_t n = 0;
51
52 /* DOS-style backslashes are accepted anywhere and normalized to '/'. */
53 const char *src = in;
54 if (*src == '/' || *src == '\\') {
55 /* Already absolute. */
56 while (*src && n < sizeof(tmp) - 1) {
57 tmp[n++] = (*src == '\\') ? '/' : *src;
58 src++;
59 }
60 tmp[n] = '\0';
61 } else {
62 /* Relative to current_dir. */
63 uint32_t cdlen = (uint32_t)strlen(current_dir);
64 for (uint32_t i = 0; i < cdlen && n < sizeof(tmp) - 1; i++) tmp[n++] = current_dir[i];
65 if (n == 0 || tmp[n - 1] != '/') {
66 if (n < sizeof(tmp) - 1) tmp[n++] = '/';
67 }
68 while (*src && n < sizeof(tmp) - 1) {
69 tmp[n++] = (*src == '\\') ? '/' : *src;
70 src++;
71 }
72 tmp[n] = '\0';
73 }
74
75 uint32_t i = 0;
76 while (tmp[i] && i < out_size - 1) { out[i] = tmp[i]; i++; }
77 out[i] = '\0';
78}

80APIRET DosSetCurrentDir(const char *dirname)
81{
82 char resolved[VFS_PATH_MAX];
83 DosResolvePath(dirname, resolved, sizeof(resolved));
84
85 vfs_stat_t st;
86 int r = vfs_stat(resolved, &st);
87 if (r != VFS_OK) return vfs_to_os2_error(r);

164/* Minimal DOS-style wildcard match: '*' = any run of chars, '?' = one char. */
165static bool wildcard_match(const char *pattern, const char *name)
166{
167 while (*pattern && *name) {
168 if (*pattern == '*') {
169 pattern++;
170 if (!*pattern) return true;
171 while (*name) {
172 if (wildcard_match(pattern, name)) return true;
173 name++;
174 }
175 return false;
176 }
177 if (*pattern == '?' || *pattern == *name) {
178 pattern++;
179 name++;
180 continue;
181 }
182 return false;
183 }
184 while (*pattern == '*') pattern++;
185 return *pattern == '\0' && *name == '\0';
186}

188/* Split "/dir/PATTERN" into dir ("/dir") and the trailing name pattern. */
189static void split_pattern(const char *path, char *dir_out, uint32_t dir_size, char *pattern_out, uint32_t pattern_size)
190{
191 int len = (int)strlen(path);
192 int slash = -1;
193 for (int i = 0; i < len; i++) {
194 if (path[i] == '/') slash = i;
195 }
196
197 if (slash < 0) {
198 strncpy(dir_out, "/", dir_size);
199 strncpy(pattern_out, path, pattern_size);
200 return;
201 }
202
203 uint32_t dlen = (uint32_t)slash;
204 if (dlen == 0) dlen = 1; /* root */
205 if (dlen >= dir_size) dlen = dir_size - 1;
206 memcpy(dir_out, path, dlen);
207 dir_out[dlen] = '\0';
208
209 const char *name = path + slash + 1;
210 strncpy(pattern_out, name, pattern_size);
211}

227/* Directory search state: HDIR is a VFS dfd plus the remembered pattern. */
228typedef struct {
229 int dfd;
230 char dir[VFS_PATH_MAX];
231 char pattern[VFS_NAME_MAX + 1];
232} find_state_t;

234#define MAX_FIND_HANDLES 16
235static find_state_t find_states[MAX_FIND_HANDLES];
236static bool find_used[MAX_FIND_HANDLES];
237
238static int alloc_find_handle(void)
239{
240 for (int i = 0; i < MAX_FIND_HANDLES; i++) {
241 if (!find_used[i]) return i;
242 }
243 return -1;
244}
245
246static APIRET find_next_locked(find_state_t *fs, FILEFINDBUF *buffer, uint32_t *count)
247{
248 vfs_dirent_t ent;
249 while (vfs_readdir(fs->dfd, &ent) == VFS_OK) {
250 if (!wildcard_match(fs->pattern, ent.name)) continue;
251
252 char full[VFS_PATH_MAX];
253 uint32_t dlen = (uint32_t)strlen(fs->dir);
254 uint32_t i = 0;
255 while (i < dlen && i < sizeof(full) - 1) { full[i] = fs->dir[i]; i++; }
256 if (i == 0 || full[i - 1] != '/') { if (i < sizeof(full) - 1) full[i++] = '/'; }
257 uint32_t j = 0;
258 while (ent.name[j] && i < sizeof(full) - 1) full[i++] = ent.name[j++];
259 full[i] = '\0';
260
261 vfs_stat_t st;
262 if (vfs_stat(full, &st) != VFS_OK) continue;
263
264 fill_findbuf(buffer, &st, ent.name);
265 if (count) *count = 1;
266 return NO_ERROR;
267 }
268 if (count) *count = 0;
269 return ERROR_NO_MORE_FILES;
270}

272APIRET DosFindFirst(const char *pattern, HDIR *handle, uint32_t attributes,
273 FILEFINDBUF *buffer, uint32_t buffer_size, uint32_t *count)
274{
275 (void)attributes; (void)buffer_size;
276 if (!handle || !buffer) return ERROR_INVALID_PARAMETER;
277
278 char resolved[VFS_PATH_MAX];
279 DosResolvePath(pattern, resolved, sizeof(resolved));
280
281 int slot = alloc_find_handle();
282 if (slot < 0) return ERROR_NOT_ENOUGH_MEMORY;
283
284 find_state_t *fs = &find_states[slot];
285 split_pattern(resolved, fs->dir, sizeof(fs->dir), fs->pattern, sizeof(fs->pattern));
286
287 int dfd = vfs_opendir(fs->dir);
288 if (dfd < 0) return vfs_to_os2_error(dfd);
289 fs->dfd = dfd;
290 find_used[slot] = true;
291
292 APIRET r = find_next_locked(fs, buffer, count);
293 if (r != NO_ERROR) {
294 vfs_closedir(fs->dfd);
295 find_used[slot] = false;
296 return r;
297 }
298
299 *handle = (HDIR)slot;
300 return NO_ERROR;
301}

303APIRET DosFindNext(HDIR handle, FILEFINDBUF *buffer, uint32_t buffer_size, uint32_t *count)
304{
305 (void)buffer_size;
306 if (handle >= MAX_FIND_HANDLES || !find_used[handle] || !buffer) return ERROR_INVALID_HANDLE;
307 return find_next_locked(&find_states[handle], buffer, count);
308}

429APIRET APIENTRY KernCopyIn(PVOID trgt, PVOID src, ULONG size)
430{
431 /* Real OS/2 validated src against the calling ring-3 process's address
432 * space (its LDT/segment limits) before copying, since kernel and user
433 * code lived in separate address spaces. This port has no MMU-backed
434 * process isolation - kernel, CMD.EXE, and everything else share one
435 * flat address space (see README.md) - so there is no separate process
436 * boundary to check src against. What *can* still be validated here,
437 * architecture-independent of any MMU: that trgt/src aren't NULL, that
438 * trgt+size/src+size don't overflow and wrap the address space (which
439 * would otherwise silently copy to/from the wrong place), and that the
440 * two regions don't overlap (memcpy() is undefined behavior on
441 * overlapping buffers; memmove() below is the safe alternative). */
442 if (!trgt || !src) return ERROR_INVALID_PARAMETER;
443 if (size == 0) return NO_ERROR;
444
445 uintptr_t trgt_addr = (uintptr_t)trgt;
446 uintptr_t src_addr = (uintptr_t)src;
447 if (trgt_addr + size < trgt_addr) return ERROR_INVALID_PARAMETER;
448 if (src_addr + size < src_addr) return ERROR_INVALID_PARAMETER;
449
450 memmove(trgt, src, size);
451 return NO_ERROR;
452}

482APIRET APIENTRY KernVMAlloc(ULONG size, ULONG flags, PVOID *linaddr, PULONG physaddr, PULONG sel)
483{
484 if (!linaddr || size == 0) return ERROR_INVALID_PARAMETER;
485
486 /* This port has one flat address space with no MMU and a simple bump
487 * allocator (mem_alloc(), kernel/src/mem.c) - no physical-vs-linear
488 * address distinction, no swapping, no GDT selectors, and no way to
489 * place an allocation at a caller-chosen or range-constrained address.
490 * Flags that are just hints in that world - FIXED/SWAP (nothing is
491 * ever swapped either way), SHARED/PROCESS (everything already shares
492 * the one address space), SGSCONT (no screen-group bookkeeping to
493 * register with), RESERVE (mem_alloc always commits immediately - a
494 * safe superset of "don't commit yet"), CONTIG (a bump allocator's
495 * output is always contiguous), and GETSEL (no GDT to return a
496 * selector from) - are accepted and effectively no-ops. Flags that
497 * promise a specific placement this allocator cannot deliver - PHYS
498 * (a caller-chosen physical address), 16MB/USEHIGHMEM (an address
499 * range guarantee) - are honestly rejected rather than silently
500 * ignored: a caller relying on e.g. sub-16MB memory for legacy DMA
501 * must not be told it succeeded when it didn't. */
502 if (flags & (KEE_VMA_PHYS | KEE_VMA_16MB | KEE_VMA_USEHIGHMEM)) {
503 return ERROR_NOT_SUPPORTED;
504 }
505
506 void *ptr = mem_alloc((size_t)size);
507 if (!ptr) return ERROR_NOT_ENOUGH_MEMORY;
508
509 *linaddr = ptr;
510 if (physaddr) *physaddr = (ULONG)(uintptr_t)ptr; /* flat: physical == linear */
511 if (sel) *sel = 0; /* no segmentation/GDT on this ARM64 port */
512 return NO_ERROR;
513}

515APIRET APIENTRY KernVMFree(PVOID linaddr)
516{
517 if (!linaddr) return ERROR_INVALID_PARAMETER;
518
519 /* mem_alloc() (kernel/src/mem.c) is a simple bump allocator with no
520 * free-list, so mem_free() is intentionally a no-op - this can't
521 * actually reclaim the memory. Still validates the pointer and
522 * returns success, same as KernVMAlloc's flag handling above: a
523 * caller checking the return code gets an honest NO_ERROR for a
524 * well-formed free, it just can't be told the memory is reusable. */
525 mem_free(linaddr);
526 return NO_ERROR;
527}

529APIRET APIENTRY KernVMLock(ULONG flags, PVOID linaddr, ULONG size,
530 KEEVMLOCK *lock, KEEVMPageList *pglist, PULONG pgsize)
531{
532 (void)flags;
533
534 if (!linaddr || size == 0) return ERROR_INVALID_PARAMETER;
535
536 /* This port has one flat address space with no MMU and nothing is
537 * ever paged out (see the comment above KernVMAlloc) - so every
538 * region is already permanently resident, and locking it against
539 * swap-out (or just verifying it's accessible, KEE_VML_VERIFYONLY)
540 * trivially succeeds for any non-NULL pointer/size. There's no real
541 * lock handle or page list to hand back either; zero them out so a
542 * caller that inspects lock/pglist/pgsize doesn't mistake
543 * uninitialized memory for real entries. */
544 if (lock) memset(lock, 0, sizeof(*lock));
545 if (pglist) memset(pglist, 0, sizeof(*pglist));
546 if (pgsize) *pgsize = 0;
547 return NO_ERROR;
548}

557 * validate the handle and clear it so a stale lock can't be reused
558 * as if it were still valid. */
559 memset(lock, 0, sizeof(*lock));
