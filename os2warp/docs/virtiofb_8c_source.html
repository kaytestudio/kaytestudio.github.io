75static bool virtio_gpu_init(void)
76{
77 uint32_t magic = virtio_read32(VIRTIO_MMIO_MAGIC);
78 if (magic != 0x74726976) { /* 'virt' in little-endian */
79 uart_puts("[FB] VirtIO GPU not found\r\n");
80 return false;
81 }
82
83 uint32_t version = virtio_read32(VIRTIO_MMIO_VERSION);
84 if (version != 2) {
85 kprintf("[FB] Unsupported VirtIO version: %u\r\n", version);
86 return false;
87 }
88
89 uint32_t device_id = virtio_read32(VIRTIO_MMIO_DEVICE_ID);
90 if (device_id != VIRTIO_ID_GPU) {
91 kprintf("[FB] Not a GPU device: %u\r\n", device_id);
92 return false;
93 }
94
95 uart_puts("[FB] VirtIO GPU detected\r\n");
96
97 /* Reset device */
98 virtio_write32(VIRTIO_MMIO_STATUS, 0);
99
100 /* Acknowledge */
101 virtio_write32(VIRTIO_MMIO_STATUS, VIRTIO_CONFIG_S_ACKNOWLEDGE);
102
103 /* Driver ready */
104 virtio_write32(VIRTIO_MMIO_STATUS,
105 VIRTIO_CONFIG_S_ACKNOWLEDGE | VIRTIO_CONFIG_S_DRIVER);
106
107 /* Features OK */
108 virtio_write32(VIRTIO_MMIO_STATUS,
109 VIRTIO_CONFIG_S_ACKNOWLEDGE |
110 VIRTIO_CONFIG_S_DRIVER |
111 VIRTIO_CONFIG_S_FEATURES_OK);
112
113 /* Driver OK */
114 virtio_write32(VIRTIO_MMIO_STATUS,
115 VIRTIO_CONFIG_S_ACKNOWLEDGE |
116 VIRTIO_CONFIG_S_DRIVER |
117 VIRTIO_CONFIG_S_FEATURES_OK |
118 VIRTIO_CONFIG_S_DRIVER_OK);
119
120 uart_puts("[FB] VirtIO GPU initialized\r\n");
121 return true;
122}

129bool fb_init(void)
130{
131 if (fb.initialized) {
132 uart_puts("[FB] Already initialized\r\n");
133 return true;
134 }
135
136 uart_puts("[FB] Initializing framebuffer...\r\n");
137
138 /* Initialize VirtIO GPU (optional - can work without it) */
139 virtio_gpu_init();
140
141 /* Allocate front buffer (16 MB aligned for performance) */
142 fb.pixels = (uint32_t *)mem_alloc(FB_SIZE);
143 if (!fb.pixels) {
144 uart_puts("[FB] ERROR: Failed to allocate front buffer\r\n");
145 return false;
146 }
147
148 /* Allocate back buffer for double buffering */
149 fb.backbuffer = (uint32_t *)mem_alloc(FB_SIZE);
150 if (!fb.backbuffer) {
151 uart_puts("[FB] WARNING: No double buffering (low memory)\r\n");
152 fb.backbuffer = fb.pixels; /* Use single buffer */
153 fb.double_buffer = false;
154 } else {
155 fb.double_buffer = true;
156 }
157
158 fb.width = FB_WIDTH;
159 fb.height = FB_HEIGHT;
160 fb.stride = FB_STRIDE;
161 fb.initialized = true;
162
163 /* Clear screen to black */
164 for (int i = 0; i < FB_WIDTH * FB_HEIGHT; i++) {
165 fb.pixels[i] = 0xFF000000; /* Black (ARGB) */
166 if (fb.double_buffer) {
167 fb.backbuffer[i] = 0xFF000000;
168 }
169 }
170
171 kprintf("[FB] Framebuffer ready: %dx%d @ %d-bit\r\n",
172 FB_WIDTH, FB_HEIGHT, FB_BPP);
173 kprintf("[FB] Front buffer: 0x%p\r\n", fb.pixels);
174 kprintf("[FB] Back buffer: 0x%p\r\n", fb.backbuffer);
175 kprintf("[FB] Double buffering: %s\r\n",
176 fb.double_buffer ? "enabled" : "disabled");
177
178 return true;
179}

290void fb_draw_line(int x0, int y0, int x1, int y1, uint32_t color)
291{
292 int dx = x1 - x0;
293 int dy = y1 - y0;
294
295 if (dx < 0) dx = -dx;
296 if (dy < 0) dy = -dy;
297
298 int sx = x0 < x1 ? 1 : -1;
299 int sy = y0 < y1 ? 1 : -1;
300 int err = dx - dy;
301
302 while (1) {
303 fb_set_pixel(x0, y0, color);
304
305 if (x0 == x1 && y0 == y1) break;
306
307 int e2 = 2 * err;
308 if (e2 > -dy) { err -= dy; x0 += sx; }
309 if (e2 < dx) { err += dx; y0 += sy; }
310 }
311}
