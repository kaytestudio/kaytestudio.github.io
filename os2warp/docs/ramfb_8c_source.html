75/* Trigger one fw_cfg DMA operation and wait for the device to finish it.
76 * `control` should already include FW_CFG_DMA_CTL_SELECT/READ/WRITE (and
77 * the target selector shifted into bits [31:16]) as needed. Returns false
78 * on a device-reported error or if it never completed (no fw_cfg DMA
79 * interface present - shouldn't happen on the "virt" machine, but a
80 * bounded retry beats an unbootable hang if this ever runs somewhere
81 * that lacks it). */
82static bool fw_cfg_dma_transfer(uint32_t control, uint32_t length, void *addr)
83{
84 volatile fw_cfg_dma_t dma;
85 dma.control = __builtin_bswap32(control);
86 dma.length = __builtin_bswap32(length);
87 dma.address = __builtin_bswap64((uint64_t)(uintptr_t)addr);
88
89 __asm__ volatile("dsb sy" ::: "memory");
90 *(volatile uint64_t *)FW_CFG_DMA_REG = __builtin_bswap64((uint64_t)(uintptr_t)&dma);
91 __asm__ volatile("dsb sy" ::: "memory");
92
93 uint32_t spins = 0;
94 while ((__builtin_bswap32(dma.control) & ~(uint32_t)FW_CFG_DMA_CTL_ERROR) != 0) {
95 if (++spins > 10000000UL) {
96 uart_puts("[FB] fw_cfg DMA timed out\r\n");
97 return false;
98 }
99 __asm__ volatile("nop");
100 }
101 return (__builtin_bswap32(dma.control) & FW_CFG_DMA_CTL_ERROR) == 0;
102}

104/* Look up a named fw_cfg file's selector key via the file directory
105 * (selector 0x19: a BE32 count followed by that many fixed-size
106 * fw_cfg_file_t entries). Returns 0 (never a valid selector for a real
107 * file) if not found. */
108static uint16_t fw_cfg_find_file(const char *name)
109{
110 static uint8_t dirbuf[4 + FW_CFG_MAX_FILES * sizeof(fw_cfg_file_t)]
111 __attribute__((aligned(8)));
112
113 uint32_t control = ((uint32_t)FW_CFG_FILE_DIR << 16) |
114 FW_CFG_DMA_CTL_SELECT | FW_CFG_DMA_CTL_READ;
115 if (!fw_cfg_dma_transfer(control, sizeof(dirbuf), dirbuf)) {
116 uart_puts("[FB] fw_cfg file directory read failed\r\n");
117 return 0;
118 }
119
120 uint32_t count = __builtin_bswap32(*(uint32_t *)dirbuf);
121 if (count > FW_CFG_MAX_FILES) count = FW_CFG_MAX_FILES;
122
123 for (uint32_t i = 0; i < count; i++) {
124 const fw_cfg_file_t *f =
125 (const fw_cfg_file_t *)(dirbuf + 4 + i * sizeof(fw_cfg_file_t));
126 if (strncmp(f->name, name, sizeof(f->name)) == 0) {
127 return __builtin_bswap16(f->select);
128 }
129 }
130 return 0;
131}

133bool ramfb_init(void *fb_addr, uint32_t width, uint32_t height, uint32_t stride)
134{
135 uint16_t sel = fw_cfg_find_file("etc/ramfb");
136 if (sel == 0) {
137 uart_puts("[FB] \"etc/ramfb\" not found - boot with -device ramfb to see graphics\r\n");
138 return false;
139 }
140
141 static ramfb_cfg_t cfg __attribute__((aligned(8)));
142 cfg.addr = __builtin_bswap64((uint64_t)(uintptr_t)fb_addr);
143 cfg.fourcc = __builtin_bswap32(DRM_FOURCC_XR24);
144 cfg.flags = 0;
145 cfg.width = __builtin_bswap32(width);
146 cfg.height = __builtin_bswap32(height);
147 cfg.stride = __builtin_bswap32(stride);
148
149 uint32_t control = ((uint32_t)sel << 16) |
150 FW_CFG_DMA_CTL_SELECT | FW_CFG_DMA_CTL_WRITE;
151 if (!fw_cfg_dma_transfer(control, sizeof(cfg), &cfg)) {
152 uart_puts("[FB] ramfb configuration write failed\r\n");
153 return false;
154 }
155
156 kprintf("[FB] ramfb configured: %ux%u @ 0x%p (QEMU will now display this buffer)\r\n",
157 width, height, fb_addr);
158 return true;
159}
