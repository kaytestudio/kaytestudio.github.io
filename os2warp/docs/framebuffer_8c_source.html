125void fb_drawstring(int x, int y, const char *str, fb_color_t fg, fb_color_t bg)
126{
127 int cx = x;
128 int cy = y;
129
130 while (*str) {
131 const uint8_t *glyph = font_8x8[(unsigned char)*str];
132
133 /* Draw character */
134 for (int row = 0; row < 8; row++) {
135 uint8_t bits = glyph[row];
136 for (int col = 0; col < 8; col++) {
137 if (bits & (1 << (7 - col))) {
138 fb_set_pixel(cx + col, cy + row, fg);
139 } else {
140 fb_set_pixel(cx + col, cy + row, bg);
141 }
142 }
143 }
144
145 cx += 8;
146 str++;
147 }
148}

153void fb_draw_button(const fb_rect_t *rect, const char *text, bool pressed)
154{
155 /* Button background */
156 fb_color_t bg_color = pressed ? FB_COLOR(160, 160, 160) : FB_COLOR(192, 192, 192);
157 fb_fillrect(rect, bg_color);
158
159 /* Border */
160 fb_color_t light = FB_COLOR(255, 255, 255);
161 fb_color_t dark = FB_COLOR(64, 64, 64);
162
163 /* Top and left - light when not pressed, dark when pressed */
164 fb_color_t top_left = pressed ? dark : light;
165 fb_color_t bottom_right = pressed ? light : dark;
166
167 /* Draw borders */
168 for (int i = 0; i < rect->width; i++) {
169 fb_set_pixel(rect->x + i, rect->y, top_left);
170 fb_set_pixel(rect->x + i, rect->y + rect->height - 1, bottom_right);
171 }
172 for (int i = 0; i < rect->height; i++) {
173 fb_set_pixel(rect->x, rect->y + i, top_left);
174 fb_set_pixel(rect->x + rect->width - 1, rect->y + i, bottom_right);
175 }
176
177 /* Center text */
178 int text_len = 0;
179 while (text[text_len]) text_len++;
180 int text_x = rect->x + (rect->width - text_len * 8) / 2;
181 int text_y = rect->y + (rect->height - 8) / 2;
182
183 fb_drawstring(text_x, text_y, text, WPS_TEXT_BLACK, bg_color);
184}
