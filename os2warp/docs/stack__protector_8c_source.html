24/* Non-zero compile-time default so protected functions are correctly
25 * protected even in the small window before stack_protector_reseed()
26 * runs (see kernel_main()) - must never be exactly 0, since an overflow
27 * that happens to zero-fill past the canary would otherwise look
28 * "unchanged" against a zero guard. */
29uintptr_t __stack_chk_guard = 0xA5A5A5A5DEADBEEFULL;

31/* Called once, early in kernel_main(), to fold in something better than
32 * a fixed constant - not a real CSPRNG (no entropy source this early in
33 * boot), just the free-running virtual counter, but enough that two
34 * boots don't share an identical, guessable canary. */
35void stack_protector_reseed(void)
36{
37 uintptr_t counter;
38 __asm__ volatile("mrs %0, cntvct_el0" : "=r"(counter));
39 __stack_chk_guard ^= counter;
40 if (__stack_chk_guard == 0) __stack_chk_guard = 0xA5A5A5A5DEADBEEFULL;
41}
