66static char *itoa_dec(int value, char *out)
67{
68 if (value == 0) {
69 out[0] = '0';
70 out[1] = '\0';
71 return out;
72 }
73
74 char tmp[12];
75 int i = 0;
76 while (value > 0 && i < (int)sizeof(tmp)) {
77 tmp[i++] = (char)('0' + (value % 10));
78 value /= 10;
79 }
80
81 int j = 0;
82 while (i > 0) {
83 out[j++] = tmp[--i];
84 }
85 out[j] = '\0';
86 return out;
87}

156static void draw_cursor(void)
157{
158 /* Simple 11x16 arrow cursor using uint16_t for proper bit width */
159 static const uint16_t cursor_mask[16] = {
160 0x400, /* 10000000000 */
161 0x600, /* 11000000000 */
162 0x700, /* 11100000000 */
163 0x780, /* 11110000000 */
164 0x7C0, /* 11111000000 */
165 0x7E0, /* 11111100000 */
166 0x7F0, /* 11111110000 */
167 0x7F8, /* 11111111000 */
168 0x7FC, /* 11111111100 */
169 0x7FE, /* 11111111110 */
170 0x7E0, /* 11111100000 */
171 0x6E0, /* 11011100000 */
172 0x430, /* 10000110000 */
173 0x030, /* 00000110000 */
174 0x018, /* 00000011000 */
175 0x000, /* 00000000000 */
176 };
177
178 /* Draw white cursor with black outline */
179 for (int y = 0; y < 16; y++) {
180 uint16_t row = cursor_mask[y];
181 for (int x = 0; x < 11; x++) {
182 if (row & (1 << (10 - x))) {
183 /* Draw white pixel */
184 fb_putpixel(mouse_x + x, mouse_y + y, WPS_TEXT_WHITE);
185 /* Draw black outline */
186 if (x > 0 && !(row & (1 << (11 - x)))) {
187 fb_putpixel(mouse_x + x - 1, mouse_y + y, WPS_TEXT_BLACK);
188 }
189 }
190 }
191 }
192}
