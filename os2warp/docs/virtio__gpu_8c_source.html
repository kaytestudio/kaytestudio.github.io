216/* Send one command as a synchronous request/response pair (descriptors
217 * 0 and 1) and busy-wait for the device to answer - see file header for
218 * why this is safe without a free-list/rotation scheme. Returns false on
219 * timeout (device wedged or never actually attached). */
220static bool gpu_ctrlq_submit(const void *cmd, uint32_t cmd_len, void *resp, uint32_t resp_len)
221{
222 g_gpu.desc[0].addr = (uint64_t)(uintptr_t)cmd;
223 g_gpu.desc[0].len = cmd_len;
224 g_gpu.desc[0].flags = VRING_DESC_F_NEXT;
225 g_gpu.desc[0].next = 1;
226
227 g_gpu.desc[1].addr = (uint64_t)(uintptr_t)resp;
228 g_gpu.desc[1].len = resp_len;
229 g_gpu.desc[1].flags = VRING_DESC_F_WRITE;
230 g_gpu.desc[1].next = 0;
231
232 uint16_t slot = (uint16_t)(g_gpu.avail.idx % QUEUE_SIZE);
233 g_gpu.avail.ring[slot] = 0;
234 __asm__ volatile("dsb sy" ::: "memory");
235 g_gpu.avail.idx++;
236 __asm__ volatile("dsb sy" ::: "memory");
237 mmio_write(g_gpu.base, VIRTIO_MMIO_QUEUE_NOTIFY, 0);
238
239 uint64_t spins = 0;
240 while (g_gpu.used.idx == g_gpu.last_used_idx) {
241 if (++spins > GPU_CTRLQ_TIMEOUT) {
242 uart_puts("[GPU] virtio-gpu: control queue request timed out\r\n");
243 return false;
244 }
245 __asm__ volatile("dsb sy" ::: "memory");
246 }
247 g_gpu.last_used_idx++;
248 return true;
249}

251static bool gpu_cmd_nodata(const void *cmd, uint32_t cmd_len, const char *what)
252{
253 virtio_gpu_ctrl_hdr_t resp;
254 if (!gpu_ctrlq_submit(cmd, cmd_len, &resp, sizeof(resp))) return false;
255 if (resp.type != VIRTIO_GPU_RESP_OK_NODATA) {
256 kprintf("[GPU] %s failed (resp type 0x%x)\r\n", what, resp.type);
257 return false;
258 }
259 return true;
260}

262void virtio_gpu_flush(int x, int y, int w, int h)
263{
264 if (!g_available) return;
265
266 virtio_gpu_transfer_to_host_2d_t xfer;
267 memset(&xfer, 0, sizeof(xfer));
268 xfer.hdr.type = VIRTIO_GPU_CMD_TRANSFER_TO_HOST_2D;
269 xfer.r.x = (uint32_t)x;
270 xfer.r.y = (uint32_t)y;
271 xfer.r.width = (uint32_t)w;
272 xfer.r.height = (uint32_t)h;
273 xfer.offset = (uint64_t)y * g_stride + (uint64_t)x * 4;
274 xfer.resource_id = GPU_RESOURCE_ID;
275 if (!gpu_cmd_nodata(&xfer, sizeof(xfer), "TRANSFER_TO_HOST_2D")) return;
276
277 virtio_gpu_resource_flush_t flush;
278 memset(&flush, 0, sizeof(flush));
279 flush.hdr.type = VIRTIO_GPU_CMD_RESOURCE_FLUSH;
280 flush.r = xfer.r;
281 flush.resource_id = GPU_RESOURCE_ID;
282 gpu_cmd_nodata(&flush, sizeof(flush), "RESOURCE_FLUSH");
283}

290bool virtio_gpu_init(void *fb_addr, uint32_t width, uint32_t height, uint32_t stride)
291{
292 if (stride != width * 4) {
293 uart_puts("[GPU] virtio-gpu: only tightly-packed (stride == width*4) buffers supported\r\n");
294 return false;
295 }
296
297 memset(&g_gpu, 0, sizeof(g_gpu));
298
299 uintptr_t base = 0;
300 bool found = false;
301 for (int i = 0; i < VIRTIO_MMIO_SLOT_COUNT; i++) {
302 uintptr_t slot = VIRTIO_MMIO_SLOT_BASE + (uintptr_t)i * VIRTIO_MMIO_SLOT_STRIDE;
303 if (mmio_read(slot, VIRTIO_MMIO_MAGIC_VALUE) != 0x74726976UL) continue; /* "virt" */
304 if (mmio_read(slot, VIRTIO_MMIO_VERSION) != 2) continue;
305 if (mmio_read(slot, VIRTIO_MMIO_DEVICE_ID) != VIRTIO_ID_GPU) continue;
306 base = slot;
307 found = true;
308 break;
309 }
310 if (!found) {
311 uart_puts("[GPU] No virtio-gpu-device found "
312 "(boot with -device virtio-gpu-device; ramfb will be used instead)\r\n");
313 return false;
314 }
315 g_gpu.base = base;
316
317 mmio_write(base, VIRTIO_MMIO_STATUS, 0);
318 mmio_write(base, VIRTIO_MMIO_STATUS, VIRTIO_CONFIG_S_ACKNOWLEDGE);
319 mmio_write(base, VIRTIO_MMIO_STATUS, VIRTIO_CONFIG_S_ACKNOWLEDGE | VIRTIO_CONFIG_S_DRIVER);
320
321 /* Negotiate only VIRTIO_F_VERSION_1 (feature bit 32 = word 1, bit 0) -
322 * in particular no VIRTIO_GPU_F_VIRGL (bit 0 of word 0) yet, and no
323 * EVENT_IDX/PACKED, keeping the plain split-ring layout above valid. */
324 mmio_write(base, VIRTIO_MMIO_DEVICE_FEATURES_SEL, 0);
325 mmio_write(base, VIRTIO_MMIO_DRIVER_FEATURES_SEL, 0);
326 mmio_write(base, VIRTIO_MMIO_DRIVER_FEATURES, 0);
327 mmio_write(base, VIRTIO_MMIO_DRIVER_FEATURES_SEL, 1);
328 mmio_write(base, VIRTIO_MMIO_DRIVER_FEATURES, 1); /* bit 0 of word 1 */
329
330 mmio_write(base, VIRTIO_MMIO_STATUS,
331 VIRTIO_CONFIG_S_ACKNOWLEDGE | VIRTIO_CONFIG_S_DRIVER | VIRTIO_CONFIG_S_FEATURES_OK);
332 if (!(mmio_read(base, VIRTIO_MMIO_STATUS) & VIRTIO_CONFIG_S_FEATURES_OK)) {
333 uart_puts("[GPU] virtio-gpu: feature negotiation rejected\r\n");
334 return false;
335 }
336
337 mmio_write(base, VIRTIO_MMIO_QUEUE_SEL, 0); /* control queue */
338 if (mmio_read(base, VIRTIO_MMIO_QUEUE_NUM_MAX) < QUEUE_SIZE) {
339 uart_puts("[GPU] virtio-gpu: control queue too small\r\n");
340 return false;
341 }
342 mmio_write(base, VIRTIO_MMIO_QUEUE_NUM, QUEUE_SIZE);
343
344 __asm__ volatile("dsb sy" ::: "memory");
345
346 uint64_t desc_addr = (uint64_t)(uintptr_t)g_gpu.desc;
347 uint64_t avail_addr = (uint64_t)(uintptr_t)&g_gpu.avail;
348 uint64_t used_addr = (uint64_t)(uintptr_t)&g_gpu.used;
349
350 mmio_write(base, VIRTIO_MMIO_QUEUE_DESC_LOW, (uint32_t)desc_addr);
351 mmio_write(base, VIRTIO_MMIO_QUEUE_DESC_HIGH, (uint32_t)(desc_addr >> 32));
352 mmio_write(base, VIRTIO_MMIO_QUEUE_DRIVER_LOW, (uint32_t)avail_addr);
353 mmio_write(base, VIRTIO_MMIO_QUEUE_DRIVER_HIGH,(uint32_t)(avail_addr >> 32));
354 mmio_write(base, VIRTIO_MMIO_QUEUE_DEVICE_LOW, (uint32_t)used_addr);
355 mmio_write(base, VIRTIO_MMIO_QUEUE_DEVICE_HIGH,(uint32_t)(used_addr >> 32));
356 mmio_write(base, VIRTIO_MMIO_QUEUE_READY, 1);
357
358 mmio_write(base, VIRTIO_MMIO_STATUS,
359 VIRTIO_CONFIG_S_ACKNOWLEDGE | VIRTIO_CONFIG_S_DRIVER |
360 VIRTIO_CONFIG_S_FEATURES_OK | VIRTIO_CONFIG_S_DRIVER_OK);
361
362 g_gpu.ready = true;
363 kprintf("[GPU] virtio-gpu control queue ready at %p\r\n", (void *)base);
364
365 /* Informational only - we still drive our own fixed width/height,
366 * not whatever the host reports here. */
367 virtio_gpu_ctrl_hdr_t get_di;
368 memset(&get_di, 0, sizeof(get_di));
369 get_di.type = VIRTIO_GPU_CMD_GET_DISPLAY_INFO;
370 virtio_gpu_resp_display_info_t di_resp;
371 if (gpu_ctrlq_submit(&get_di, sizeof(get_di), &di_resp, sizeof(di_resp)) &&
372 di_resp.hdr.type == VIRTIO_GPU_RESP_OK_DISPLAY_INFO) {
373 kprintf("[GPU] host scanout0: %ux%u enabled=%u\r\n",
374 di_resp.pmodes[0].r.width, di_resp.pmodes[0].r.height,
375 di_resp.pmodes[0].enabled);
376 }
377
378 virtio_gpu_resource_create_2d_t create;
379 memset(&create, 0, sizeof(create));
380 create.hdr.type = VIRTIO_GPU_CMD_RESOURCE_CREATE_2D;
381 create.resource_id = GPU_RESOURCE_ID;
382 create.format = VIRTIO_GPU_FORMAT_B8G8R8X8_UNORM;
383 create.width = width;
384 create.height = height;
385 if (!gpu_cmd_nodata(&create, sizeof(create), "RESOURCE_CREATE_2D")) return false;
386
387 virtio_gpu_resource_attach_backing_1_t attach;
388 memset(&attach, 0, sizeof(attach));
389 attach.hdr.type = VIRTIO_GPU_CMD_RESOURCE_ATTACH_BACKING;
390 attach.resource_id = GPU_RESOURCE_ID;
391 attach.nr_entries = 1;
392 attach.entry.addr = (uint64_t)(uintptr_t)fb_addr;
393 attach.entry.length = stride * height;
394 if (!gpu_cmd_nodata(&attach, sizeof(attach), "RESOURCE_ATTACH_BACKING")) return false;
395
396 virtio_gpu_set_scanout_t scan;
397 memset(&scan, 0, sizeof(scan));
398 scan.hdr.type = VIRTIO_GPU_CMD_SET_SCANOUT;
399 scan.r.x = 0;
400 scan.r.y = 0;
401 scan.r.width = width;
402 scan.r.height = height;
403 scan.scanout_id = GPU_SCANOUT_ID;
404 scan.resource_id = GPU_RESOURCE_ID;
405 if (!gpu_cmd_nodata(&scan, sizeof(scan), "SET_SCANOUT")) return false;
406
407 g_stride = stride;
408 g_available = true;
409 uart_puts("[GPU] virtio-gpu 2D scanout ready\r\n");
410
411 virtio_gpu_flush(0, 0, (int)width, (int)height);
412 return true;
413}
