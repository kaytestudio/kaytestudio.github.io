171/* Query whether this device advertises any EV_ABS axes: plain
172 * synchronous config-space access (offset 0x100+), no DMA/barriers
173 * needed (see file header). A keyboard reports size=0 (no absolute
174 * axes); a tablet reports size>0. */
175static bool device_has_ev_abs(uintptr_t base)
176{
177 mmio_write8(base, VIRTIO_MMIO_CONFIG + 0, VIRTIO_INPUT_CFG_EV_BITS); /* select */
178 mmio_write8(base, VIRTIO_MMIO_CONFIG + 1, EV_ABS); /* subsel */
179 uint8_t size = mmio_read8(base, VIRTIO_MMIO_CONFIG + 2);
180 return size > 0;
181}

183/* Hand descriptor `idx` back to the device as an available (writable)
184 * receive buffer, so the next event can be delivered into it. */
185static void requeue_buffer(vinput_dev_t *dev, uint16_t idx)
186{
187 uint16_t slot = (uint16_t)(dev->avail.idx % QUEUE_SIZE);
188 dev->avail.ring[slot] = idx;
189 __asm__ volatile("dsb sy" ::: "memory");
190 dev->avail.idx++;
191 __asm__ volatile("dsb sy" ::: "memory");
192 mmio_write(dev->base, VIRTIO_MMIO_QUEUE_NOTIFY, 0);
193}

195/* Standard virtio-mmio handshake + one receive virtqueue, shared by
196 * keyboard and tablet setup. */
197static bool vinput_dev_setup(vinput_dev_t *dev, uintptr_t base)
198{
199 memset(dev, 0, sizeof(*dev));
200 dev->base = base;
201
202 mmio_write(base, VIRTIO_MMIO_STATUS, 0);
203 mmio_write(base, VIRTIO_MMIO_STATUS, VIRTIO_CONFIG_S_ACKNOWLEDGE);
204 mmio_write(base, VIRTIO_MMIO_STATUS, VIRTIO_CONFIG_S_ACKNOWLEDGE | VIRTIO_CONFIG_S_DRIVER);
205
206 /* Negotiate only VIRTIO_F_VERSION_1 (feature bit 32 = word 1, bit 0);
207 * reject every other offered feature (in particular no EVENT_IDX, no
208 * PACKED ring) to keep the split-ring layout above valid. */
209 mmio_write(base, VIRTIO_MMIO_DEVICE_FEATURES_SEL, 0);
210 mmio_write(base, VIRTIO_MMIO_DRIVER_FEATURES_SEL, 0);
211 mmio_write(base, VIRTIO_MMIO_DRIVER_FEATURES, 0);
212 mmio_write(base, VIRTIO_MMIO_DRIVER_FEATURES_SEL, 1);
213 mmio_write(base, VIRTIO_MMIO_DRIVER_FEATURES, 1); /* bit 0 of word 1 */
214
215 mmio_write(base, VIRTIO_MMIO_STATUS,
216 VIRTIO_CONFIG_S_ACKNOWLEDGE | VIRTIO_CONFIG_S_DRIVER | VIRTIO_CONFIG_S_FEATURES_OK);
217 if (!(mmio_read(base, VIRTIO_MMIO_STATUS) & VIRTIO_CONFIG_S_FEATURES_OK)) {
218 uart_puts("[KBD] virtio-input: feature negotiation rejected\r\n");
219 return false;
220 }
221
222 mmio_write(base, VIRTIO_MMIO_QUEUE_SEL, 0);
223 if (mmio_read(base, VIRTIO_MMIO_QUEUE_NUM_MAX) < QUEUE_SIZE) {
224 uart_puts("[KBD] virtio-input: eventq too small\r\n");
225 return false;
226 }
227 mmio_write(base, VIRTIO_MMIO_QUEUE_NUM, QUEUE_SIZE);
228
229 for (int i = 0; i < QUEUE_SIZE; i++) {
230 dev->desc[i].addr = (uint64_t)(uintptr_t)&dev->bufs[i];
231 dev->desc[i].len = sizeof(virtio_input_event_t);
232 dev->desc[i].flags = VRING_DESC_F_WRITE; /* device writes the event into it */
233 dev->avail.ring[i] = (uint16_t)i;
234 }
235 dev->avail.idx = QUEUE_SIZE; /* every buffer available up front */
236 dev->last_used_idx = 0;
237
238 __asm__ volatile("dsb sy" ::: "memory");
239
240 uint64_t desc_addr = (uint64_t)(uintptr_t)dev->desc;
241 uint64_t avail_addr = (uint64_t)(uintptr_t)&dev->avail;
242 uint64_t used_addr = (uint64_t)(uintptr_t)&dev->used;
243
244 mmio_write(base, VIRTIO_MMIO_QUEUE_DESC_LOW, (uint32_t)desc_addr);
245 mmio_write(base, VIRTIO_MMIO_QUEUE_DESC_HIGH, (uint32_t)(desc_addr >> 32));
246 mmio_write(base, VIRTIO_MMIO_QUEUE_DRIVER_LOW, (uint32_t)avail_addr);
247 mmio_write(base, VIRTIO_MMIO_QUEUE_DRIVER_HIGH, (uint32_t)(avail_addr >> 32));
248 mmio_write(base, VIRTIO_MMIO_QUEUE_DEVICE_LOW, (uint32_t)used_addr);
249 mmio_write(base, VIRTIO_MMIO_QUEUE_DEVICE_HIGH, (uint32_t)(used_addr >> 32));
250 mmio_write(base, VIRTIO_MMIO_QUEUE_READY, 1);
251
252 mmio_write(base, VIRTIO_MMIO_STATUS,
253 VIRTIO_CONFIG_S_ACKNOWLEDGE | VIRTIO_CONFIG_S_DRIVER |
254 VIRTIO_CONFIG_S_FEATURES_OK | VIRTIO_CONFIG_S_DRIVER_OK);
255
256 mmio_write(base, VIRTIO_MMIO_QUEUE_NOTIFY, 0);
257
258 dev->ready = true;
259 return true;
260}

262/* Drain at most one pending event from `dev` into *out. Returns false
263 * when the used ring is empty (nothing pending right now). */
264static bool vinput_dev_poll(vinput_dev_t *dev, virtio_input_event_t *out)
265{
266 if (!dev->ready) return false;
267 if (dev->used.idx == dev->last_used_idx) return false;
268
269 uint16_t slot = (uint16_t)(dev->last_used_idx % QUEUE_SIZE);
270 uint32_t desc_id = dev->used.ring[slot].id;
271 dev->last_used_idx++;
272
273 if (desc_id >= QUEUE_SIZE) return false; /* corrupt id; drop */
274
275 *out = dev->bufs[desc_id];
276 requeue_buffer(dev, (uint16_t)desc_id);
277 return true;
278}

280void vinput_init(void)
281{
282 bool kbd_found = false, tablet_found = false;
283
284 for (int i = 0; i < VIRTIO_MMIO_SLOT_COUNT && !(kbd_found && tablet_found); i++) {
285 uintptr_t base = VIRTIO_MMIO_SLOT_BASE + (uintptr_t)i * VIRTIO_MMIO_SLOT_STRIDE;
286 if (mmio_read(base, VIRTIO_MMIO_MAGIC_VALUE) != 0x74726976UL) continue; /* "virt" */
287 if (mmio_read(base, VIRTIO_MMIO_VERSION) != 2) continue;
288 if (mmio_read(base, VIRTIO_MMIO_DEVICE_ID) != VIRTIO_ID_INPUT) continue;
289
290 if (device_has_ev_abs(base)) {
291 if (tablet_found) continue;
292 if (vinput_dev_setup(&g_tablet, base)) {
293 tablet_found = true;
294 kprintf("[KBD] virtio-tablet ready at %p\r\n", (void *)base);
295 }
296 } else {
297 if (kbd_found) continue;
298 if (vinput_dev_setup(&g_keyboard, base)) {
299 kbd_found = true;
300 kprintf("[KBD] virtio-keyboard ready at %p\r\n", (void *)base);
301 }
302 }
303 }
304
305 if (!kbd_found) {
306 uart_puts("[KBD] No virtio-keyboard found "
307 "(boot with -device virtio-keyboard-device for Cmd+Escape)\r\n");
308 }
309 if (!tablet_found) {
310 uart_puts("[KBD] No virtio-tablet found "
311 "(boot with -device virtio-tablet-device for a mouse cursor)\r\n");
312 }
313}

315bool vinput_poll_workplace_hotkey(void)
316{
317 bool hotkey = false;
318 virtio_input_event_t ev;
319
320 while (vinput_dev_poll(&g_keyboard, &ev)) {
321 if (ev.type != EV_KEY) continue;
322 if (ev.code == KEY_LEFTMETA || ev.code == KEY_RIGHTMETA) {
323 g_meta_down = (ev.value != 0); /* 1=press, 2=repeat, 0=release */
324 } else if (ev.code == KEY_ESC && ev.value == 1 && g_meta_down) {
325 hotkey = true;
326 }
327 }
328
329 return hotkey;
330}

332bool vinput_poll_pointer(void)
333{
334 static int abs_x = (TABLET_ABS_MAX - TABLET_ABS_MIN) / 2;
335 static int abs_y = (TABLET_ABS_MAX - TABLET_ABS_MIN) / 2;
336
337 virtio_input_event_t ev;
338 bool moved = false;
339 bool changed = false;
340
341 while (vinput_dev_poll(&g_tablet, &ev)) {
342 if (ev.type == EV_ABS) {
343 if (ev.code == ABS_X) { abs_x = (int)ev.value; moved = true; }
344 else if (ev.code == ABS_Y) { abs_y = (int)ev.value; moved = true; }
345 } else if (ev.type == EV_KEY && (ev.code == BTN_LEFT || ev.code == BTN_RIGHT)) {
346 int sx = fb_get_width() * (abs_x - TABLET_ABS_MIN) / (TABLET_ABS_MAX - TABLET_ABS_MIN);
347 int sy = fb_get_height() * (abs_y - TABLET_ABS_MIN) / (TABLET_ABS_MAX - TABLET_ABS_MIN);
348 wl_seat_pointer_button(sx, sy, (ev.code == BTN_LEFT) ? 0 : 1, ev.value != 0);
349 changed = true;
350 }
351 }
352
353 if (moved) {
354 int sx = fb_get_width() * (abs_x - TABLET_ABS_MIN) / (TABLET_ABS_MAX - TABLET_ABS_MIN);
355 int sy = fb_get_height() * (abs_y - TABLET_ABS_MIN) / (TABLET_ABS_MAX - TABLET_ABS_MIN);
356 wl_seat_pointer_motion(sx, sy);
357 changed = true;
358 }
359
360 return changed;
361}
