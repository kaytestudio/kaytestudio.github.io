60static void crc32_init(void)
61{
62 for (uint32_t i = 0; i < 256; i++) {
63 uint32_t c = i;
64 for (int k = 0; k < 8; k++) {
65 c = (c & 1) ? (0xEDB88320u ^ (c >> 1)) : (c >> 1);
66 }
67 crc32_table[i] = c;
68 }
69 crc32_ready = 1;
70}

72static uint32_t crc32_calc(const uint8_t *data, uint32_t len)
73{
74 if (!crc32_ready) crc32_init();
75 uint32_t c = 0xFFFFFFFFu;
76 for (uint32_t i = 0; i < len; i++) {
77 c = crc32_table[(c ^ data[i]) & 0xFF] ^ (c >> 8);
78 }
79 return c ^ 0xFFFFFFFFu;
80}

94{
95 char tmp[12];

98 while (val > 0) { tmp[n++] = (char)('0' + (val % 10)); val /= 10; }
99 int j = 0;
100 while (n > 0) dst[j++] = tmp[--n];

104void gpt_probe(blkdev_t *disk)
105{
106 uint8_t sector[512];
107
108 if (disk->sector_size == 0 || disk->sector_size > sizeof(sector)) return;
109
110 /* Protective MBR at LBA 0: just check the 0x55AA boot signature. */
111 if (blk_read(disk, 0, 1, sector) != 0) return;
112 if (sector[510] != 0x55 || sector[511] != 0xAA) return;
113
114 /* GPT header at LBA 1. */
115 if (blk_read(disk, GPT_HEADER_LBA, 1, sector) != 0) return;
116 gpt_header_t hdr;
117 memcpy(&hdr, sector, sizeof(hdr));
118 if (memcmp(hdr.signature, GPT_SIGNATURE, 8) != 0) return;
119 if (hdr.header_size < sizeof(hdr) || hdr.header_size > sizeof(sector)) return;
120
121 uint8_t hdr_copy[512];
122 memcpy(hdr_copy, sector, sizeof(hdr_copy));
123 memset(hdr_copy + 16, 0, 4); /* zero header_crc32 field before recomputing */
124 uint32_t calc_crc = crc32_calc(hdr_copy, hdr.header_size);
125 if (calc_crc != hdr.header_crc32) {
126 kprintf("[GPT] WARNING: header CRC32 mismatch (continuing anyway)\n");
127 }
128
129 uint32_t num_entries = hdr.num_partition_entries;
130 uint32_t entry_size = hdr.partition_entry_size;
131 if (num_entries > GPT_MAX_PARTITIONS) num_entries = GPT_MAX_PARTITIONS;
132 if (entry_size == 0 || entry_size > sizeof(sector)) return;
133
134 uint32_t entries_per_sector = disk->sector_size / entry_size;
135 if (entries_per_sector == 0) return;
136
137 int found = 0;
138 for (uint32_t i = 0; i < num_entries; i++) {
139 uint32_t sector_off = i / entries_per_sector;
140 uint32_t in_sector_off = (i % entries_per_sector) * entry_size;
141
142 uint8_t entry_sector[512];
143 if (blk_read(disk, (int64_t)(hdr.partition_entry_lba + sector_off), 1, entry_sector) != 0) break;
144
145 gpt_entry_t e;
146 memcpy(&e, entry_sector + in_sector_off, sizeof(e));
147
148 int all_zero = 1;
149 for (int b = 0; b < 16; b++) {
150 if (e.type_guid[b] != 0) { all_zero = 0; break; }
151 }
152 if (all_zero) continue;
153 if (e.last_lba < e.first_lba) continue;
154
155 char name[32];
156 utf16le_to_ascii(e.name, 36, name, sizeof(name));
157 if (name[0] == '\0') {
158 strcpy(name, "part");
159 put_uint(name + 4, (unsigned)i);
160 }
161
162 blkdev_t *part = (blkdev_t *)mem_alloc(sizeof(blkdev_t));
163 if (!part) break;
164 memset(part, 0, sizeof(*part));
165
166 strncpy(part->name, name, sizeof(part->name) - 1);
167 part->start_sector = (int64_t)e.first_lba;
168 part->num_sectors = (int64_t)(e.last_lba - e.first_lba + 1);
169 part->sector_size = disk->sector_size;
170 part->read = disk->read;
171 part->write = disk->write;
172 part->driver_data = disk->driver_data;
173 part->parent = disk;
174 part->flags = BLKDEV_PARTITION;
175
176 blkdev_register(part);
177 found++;
178 }
179
180 kprintf("[GPT] Found %d partition(s) on %s\n", found, disk->name);
181}
